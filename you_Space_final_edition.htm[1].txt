class Audio {
  constructor() { this.ctx = null; this.master = 0.14; }
  init() { if (this.ctx) return; try { this.ctx = new (window.AudioContext||window.webkitAudioContext)(); } catch(e){} }
  t(freq, dur, vol, type, delay) {
    if (!this.ctx) return;
    const now = this.ctx.currentTime + (delay||0);
    const osc = this.ctx.createOscillator();
    const g = this.ctx.createGain();
    osc.type = type || 'sine';
    osc.frequency.setValueAtTime(freq, now);
    g.gain.setValueAtTime(0.0001, now);
    g.gain.exponentialRampToValueAtTime(vol*this.master, now+0.008);
    g.gain.exponentialRampToValueAtTime(0.0001, now+dur);
    osc.connect(g); g.connect(this.ctx.destination);
    osc.start(now); osc.stop(now+dur+0.02);
  }
  tick() { this.t(1400, 0.02, 0.3, 'sine', 0); }
  arrive() { this.t(660, 0.06, 0.45, 'sine', 0); this.t(990, 0.10, 0.4, 'sine', 0.06); }

let lastFrame = performance.now();

function loop(now) {
  const dt = Math.min(64, now - lastFrame);
  lastFrame = now;

  if (paused) {
    const remaining = pauseUntil - now;
    if (remaining <= 0) {
      paused = false;
      pauseRing.classList.remove('active');
      pauseRingCircle.setAttribute('stroke-dashoffset', String(PAUSE_RING_CIRC));
      audio.tick();
    } else {
      const progress = 1 - remaining / PAUSE_MS;
      pauseRing.classList.add('active');
      pauseRingCircle.setAttribute('stroke-dashoffset', String(PAUSE_RING_CIRC * (1 - progress)));
    }
    positionPlanets();
    requestAnimationFrame(loop);
    return;
  }

  if (mode === 'tour') {
    orbitAngleDeg -= TOUR_SPEED * (dt / 1000);
    rMotion.textContent = 'Tour';
    if (orbitAngleDeg <= tourEndAngle) {
      orbitAngleDeg = tourEndAngle;
      mode = 'auto';
      lastSlot = Math.floor(-orbitAngleDeg / STEP);
      demoBadge.classList.remove('on');
      audio.tourEnd();
      showToast('Tour complete');
      rMode.textContent = 'Auto';
    }
  } else {
    orbitAngleDeg -= AUTO_SPEED * (dt / 1000);
    rMotion.textContent = 'Spin';

    const slot = Math.floor(-orbitAngleDeg / STEP);
    if (slot !== lastSlot) {
      lastSlot = slot;
      paused = true;
      pauseUntil = now + PAUSE_MS;
      audio.arrive();
      updateFrontDisplay(slot);
    }
  }

  positionPlanets();
  updateFrontDisplay(Math.floor(-orbitAngleDeg / STEP));
  requestAnimationFrame(loop);
}

const HAND_CONN = [

function drawSkeleton(hand) {
  const w = handCanvas.width, h = handCanvas.height;
  const style = getComputedStyle(document.documentElement);
  const bg = style.getPropertyValue('--surface').trim();
  const accent = style.getPropertyValue('--accent').trim();
  const dim = style.getPropertyValue('--text-3').trim();

  handCtx.fillStyle = bg;
  handCtx.fillRect(0, 0, w, h);

  if (!hand) {
    handCtx.fillStyle = dim;
    handCtx.font = '600 12px monospace';
    handCtx.textAlign = 'center';
    handCtx.fillText('NO HAND', w/2, h/2 + 4);
    return;
  }

  const pts = hand.map(p => ({ x: (1 - p.x) * w, y: p.y * h }));

  handCtx.strokeStyle = accent;
  handCtx.globalAlpha = 0.7;
  handCtx.lineWidth = 2.5;
  handCtx.lineCap = 'round';
  handCtx.lineJoin = 'round';
  HAND_CONN.forEach(([a,b]) => {
    handCtx.beginPath();
    handCtx.moveTo(pts[a].x, pts[a].y);
    handCtx.lineTo(pts[b].x, pts[b].y);
    handCtx.stroke();
  });

  handCtx.globalAlpha = 1;
  pts.forEach((p, i) => {
    handCtx.fillStyle = (i === 4 || i === 8) ? accent : dim;
    handCtx.beginPath();
    handCtx.arc(p.x, p.y, (i === 4 || i === 8) ? 5 : 3, 0, Math.PI * 2);
    handCtx.fill();
  });
}

function isFingerExtended(hand, tip, pip) {
  const w = hand[0];
  return Math.hypot(hand[tip].x - w.x, hand[tip].y - w.y) >
         Math.hypot(hand[pip].x - w.x, hand[pip].y - w.y) * 1.12;
}

function countExtended(hand) {
  return [8,12,16,20].map((t, i) => isFingerExtended(hand, t, [6,10,14,18][i])).filter(Boolean).length;
}

function isThumbsUp(hand) {
  if (countExtended(hand) !== 0) return false;
  const tt = hand[4], ti = hand[3], tm = hand[2], w = hand[0];
  const up = tt.y < ti.y && ti.y < tm.y && tt.y < w.y - 0.04;
  const dx = tt.x - tm.x, dy = tt.y - tm.y;
  return up && Math.abs(dy) > Math.abs(dx) * 1.15;
}

function isVictory(hand) {
  const idx = isFingerExtended(hand, 8, 6);
  const mid = isFingerExtended(hand, 12, 10);
  const rng = isFingerExtended(hand, 16, 14);
  const pnk = isFingerExtended(hand, 20, 18);
  return idx && mid && !rng && !pnk;
}

let thumbUpActive = false;
let thumbUpStart = 0;
const THUMB_HOLD_MS = 800;
let lastThumbFired = 0;

let victoryActive = false;
let victoryStart = 0;
const VICTORY_HOLD_MS = 500;
let lastVictoryFired = 0;

function processHand(hand) {
  const fingers = countExtended(hand);
  const thumbUp = isThumbsUp(hand);
  const victory = isVictory(hand);

  handCount.textContent = fingers + ' finger' + (fingers === 1 ? '' : 's');

  if (thumbUp) {
    if (!thumbUpActive) { thumbUpActive = true; thumbUpStart = performance.now(); }
    const held = performance.now() - thumbUpStart;
    const p = Math.min(held / THUMB_HOLD_MS, 1);
    pauseRing.classList.add('active');
    pauseRingCircle.setAttribute('stroke-dashoffset', String(PAUSE_RING_CIRC * (1 - p)));
    document.querySelectorAll('.guide-item').forEach(el =>
      el.classList.toggle('active', el.dataset.g === 'THUMB_UP'));
    if (p >= 1 && performance.now() - lastThumbFired > 1200) {
      lastThumbFired = performance.now();
      thumbUpActive = false;
      pauseRing.classList.remove('active');
      pauseRingCircle.setAttribute('stroke-dashoffset', String(PAUSE_RING_CIRC));
      openFront();
    }
    return;
  } else {
    thumbUpActive = false;
    if (!paused) {
      pauseRingCircle.setAttribute('stroke-dashoffset', String(PAUSE_RING_CIRC));
      pauseRing.classList.remove('active');
    }
  }

  if (victory) {
    if (!victoryActive) { victoryActive = true; victoryStart = performance.now(); }
    const held = performance.now() - victoryStart;
    document.querySelectorAll('.guide-item').forEach(el =>
      el.classList.toggle('active', el.dataset.g === 'VICTORY'));
    if (held >= VICTORY_HOLD_MS && performance.now() - lastVictoryFired > 1500) {
      lastVictoryFired = performance.now();
      victoryActive = false;
      startTour();
    }
    return;
  } else {
    victoryActive = false;
    document.querySelectorAll('.guide-item').forEach(el =>
      el.classList.remove('active'));
  }
}

let handEnabled = false;
let handReady = false;
let handLandmarker = null;
let lastVideoTime = -1;
let lastDetectTime = 0;
const DETECT_INTERVAL = 24;

async function startHand() {
  audio.init();
  if (handReady) {
    handEnabled = true;
    handPreview.classList.add('on');
    handBtn.classList.add('on');
    handLabel.textContent = 'Hand on';
    audio.handOn();
    showToast('Hand mode active');
    return;
  }

  handBtn.classList.add('loading');
  handLabel.textContent = 'Loading';
  handLabelPrev.textContent = 'Loading';

  try {
    const vision = await import('https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/+esm');
    const fileset = await vision.FilesetResolver.forVisionTasks(
      'https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/wasm'
    );
    handLandmarker = await vision.HandLandmarker.createFromOptions(fileset, {
      baseOptions: {
        modelAssetPath: 'https://storage.googleapis.com/mediapipe-models/hand_landmarker/hand_landmarker/float16/1/hand_landmarker.task',
        delegate: 'GPU'
      },
      runningMode: 'VIDEO',
      numHands: 1
    });

    const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user' } });
    camVideo.srcObject = stream;
    await camVideo.play();

    handReady = true;
    handEnabled = true;
    handBtn.classList.remove('loading');
    handBtn.classList.add('on');
    handLabel.textContent = 'Hand on';
    handLabelPrev.textContent = 'Waiting';
    handPreview.classList.add('on');
    audio.handOn();
    showToast('Hand mode active');
    runHandLoop();
  } catch (err) {
    console.error(err);
    handBtn.classList.remove('loading');
    handLabel.textContent = 'Hand off';
    handLabelPrev.textContent = 'Error';
    showToast('Camera unavailable');
    setTimeout(() => handPreview.classList.remove('on'), 1800);
  }
}

function stopHand() {
  handEnabled = false;
  handPreview.classList.remove('on');
  handBtn.classList.remove('on');
  handLabel.textContent = 'Hand off';
  thumbUpActive = false;
  victoryActive = false;
  if (!paused) {
    pauseRing.classList.remove('active');
    pauseRingCircle.setAttribute('stroke-dashoffset', String(PAUSE_RING_CIRC));
  }
  document.querySelectorAll('.guide-item').forEach(el => el.classList.remove('active'));
  audio.handOff();
  showToast('Hand mode off');
}

function runHandLoop() {
  requestAnimationFrame(runHandLoop);
  if (!handEnabled || !handLandmarker) return;

  const now = performance.now();
  if (now - lastDetectTime < DETECT_INTERVAL) return;
  lastDetectTime = now;

  if (camVideo.readyState < 2) return;
  if (camVideo.currentTime === lastVideoTime) return;
  lastVideoTime = camVideo.currentTime;

  const result = handLandmarker.detectForVideo(camVideo, now);
  const landmarks = result.landmarks || [];

  if (landmarks.length > 0) {
    processHand(landmarks[0]);
    drawSkeleton(landmarks);
    handLabelPrev.textContent = 'Tracking';
  } else {
    drawSkeleton(null);
    thumbUpActive = false;
    victoryActive = false;
    handLabelPrev.textContent = 'No hand';
    handCount.textContent = '0 fingers';
  }
}
